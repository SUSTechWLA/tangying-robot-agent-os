type markerReviewRepository struct {tasks.Repository;armed atomic.Bool;entered,release chan struct{}}
func(r *markerReviewRepository)Get(ctx context.Context,id string)(*tasks.Task,error){task,err:=r.Repository.Get(ctx,id);if _,bounded:=ctx.Deadline();bounded&&r.armed.CompareAndSwap(true,false){close(r.entered);<-r.release};return task,err}
